}

/** シートの形を今の版にそろえる（前の版のデータは、列の名前で移しかえる） */
function ensureAll_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(SCHEMA).forEach(name => migrateSheet_(ss, name, SCHEMA[name].fields.map(f => f.key)));
  mergeOldRoutines_(ss);
  const st = ss.getSheetByName('設定') || ss.insertSheet('設定');
  st.getRange(1, 1, Math.max(st.getMaxRows(), 2), 2).setNumberFormat('@');
  st.getRange(1, 1, 1, 2).setValues([['項目', '値']]).setFontWeight('bold');
  st.setFrozenRows(1);
  const cur = readSettings_();
  // 前の版の「朝の通知・前夜の通知」から、毎日のリマインドへ移す
  if (!('リマインドの時刻' in cur)) {
    let t = '07:00', next = '';
    if (cur['朝の通知'] === '○' && cur['朝の時刻']) t = cur['朝の時刻'];
    else if (cur['前夜の通知'] === '○' && cur['前夜の時刻']) { t = cur['前夜の時刻']; next = '○'; }
    st.appendRow(['リマインドの時刻', t]);
    st.appendRow(['翌日の予定を送る', next]);
  }
  const now = readSettings_();
  Object.keys(SETTING_DEFAULTS).forEach(k => { if (!(k in now)) st.appendRow([k, SETTING_DEFAULTS[k]]); });
  PropertiesService.getScriptProperties().setProperty('SCHEMA_VER', VERSION);
}

function migrateSheet_(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  const lastCol = Math.max(sh.getLastColumn(), 1), lastRow = sh.getLastRow();
  const old = lastRow >= 1 ? sh.getRange(1, 1, lastRow, lastCol).getDisplayValues() : [];
  const oldHead = (old[0] || []).map(h => RENAMES[h] || h);
  const same = oldHead.length === headers.length && oldHead.every((h, i) => h === headers[i]);
  if (!same) {
    const rows = old.slice(1).filter(r => r.some(v => v !== '')).map(r => {
      const o = {};
      oldHead.forEach((h, i) => { o[h] = r[i]; });
      Object.keys(VALUE_RENAMES).forEach(k => { if (o[k] && VALUE_RENAMES[k][o[k]]) o[k] = VALUE_RENAMES[k][o[k]]; });
      if (name !== 'カテゴリ' && 'リマインド' in o === false) o['リマインド'] = '';
      return headers.map(h => (o[h] === undefined ? (h === '何日間' ? '1' : h === '指定のしかた' ? '日付で' : '') : o[h]));
    });
    sh.clear();
    sh.getRange(1, 1, Math.max(sh.getMaxRows(), rows.length + 2), headers.length).setNumberFormat('@');
    sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold');
    if (rows.length) sh.getRange(2, 1, rows.length, headers.length).setValues(rows);
  } else {
    sh.getRange(1, 1, Math.max(sh.getMaxRows(), 2), headers.length).setNumberFormat('@');
    sh.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  }
  sh.setFrozenRows(1);
  return sh;
}

/** 前の版の「毎日・毎週・毎月」を「定例」に移し、元のシートは消す（前の版で残した「（移す前）〇〇」も消す） */
function mergeOldRoutines_(ss) {
  const sh = ss.getSheetByName('定例');
  const headers = SCHEMA['定例'].fields.map(f => f.key);
  OLD_ROUTINES.forEach(name => {
    const old = ss.getSheetByName(name);
    if (!old) return;
    const rows = readSheet_(name).map(r => {
      const o = { ID: r['ID'] || newId_(), '種類': '予定', '内容': r['内容'], '間隔': '1', '開始': r['開始'] || r['時刻'] || '', '終了': r['終了'] || '',
        '何日間': r['何日間'] || '1', '場所': r['場所'] || '', 'カテゴリ': r['カテゴリ'] || '', 'リマインド': r['リマインド'] || '', '有効': r['有効'] };
      if (name === '毎日') {
        o['種類'] = r['種類'] === '時間の確保' ? '時間の確保' : '予定';
        if (r['曜日指定']) { o['単位'] = '週'; o['曜日'] = r['曜日指定']; } else o['単位'] = '日';
      } else if (name === '毎週') {
        o['単位'] = '週'; o['曜日'] = r['曜日'];
      } else {
        o['単位'] = 'か月'; o['月の決め方'] = r['指定のしかた'] || '日付で'; o['日'] = r['日'] || '';
        o['第何週'] = r['週'] || ''; o['何曜日'] = r['指定のしかた'] === '第〇週の曜日で' ? (r['曜日'] || '') : '';
      }
      return headers.map(h => (o[h] === undefined || o[h] === null ? '' : String(o[h])));
    });
    if (rows.length) sh.getRange(sh.getLastRow() + 1, 1, rows.length, headers.length).setValues(rows);
    ss.deleteSheet(old);
  });
  ss.getSheets().filter(x => /^（移す前）/.test(x.getName())).forEach(x => ss.deleteSheet(x));
}

function doGet() {
  return ContentService.createTextOutput('LINE秘書は動いています（' + VERSION + '）');
}

function doPost(e) {
  let res;
  try {
    const req = JSON.parse(e.postData.contents || '{}');
    res = handle_(req);
  } catch (err) {
    res = { ok: false, error: String(err && err.message ? err.message : err) };
  }
  return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
}

function handle_(req) {
  const props = PropertiesService.getScriptProperties();
  const stored = props.getProperty('KEY_HASH');
  const given = String(req.key || '');
  if (given.length < 4) return { ok: false, error: 'パスワードは4文字以上にしてください' };
  if (!stored) {
    if (req.action !== 'hello') return { ok: false, error: '先に接続してください' };
    props.setProperty('KEY_HASH', hash_(given)); // 最初につないだパスワードを登録
  } else if (stored !== hash_(given)) {
    return { ok: false, error: 'このスプレッドシートのパスワードがちがいます' };
  }

  // 利用者IDとのひもづけ（1つのスプレッドシートは1つの利用者IDだけ）
  const acc = String(req.accountId || '').trim();
  const bound = props.getProperty('ACCOUNT_ID');
  if (req.action === 'hello') {
    if (!acc) return { ok: false, error: '利用者IDがありません' };
    if (bound && bound !== acc) return { ok: false, error: 'このスプレッドシートは、別の利用者IDで使われています' };
    if (!bound) props.setProperty('ACCOUNT_ID', acc);
    CacheService.getScriptCache().remove('LIC');
    return { ok: true, version: VERSION, sheetId: SpreadsheetApp.getActiveSpreadsheet().getId() };
  }
  const lic = licenseCheck_();
  if (!lic.ok) return { ok: false, licenseError: true, error: lic.error };

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    if (props.getProperty('SCHEMA_VER') !== VERSION) { ensureAll_(); scheduleAll_(); }
    switch (req.action) {
      case 'load':
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        return { ok: true, version: VERSION, schema: SCHEMA, offsets: OFFSETS, data: readAll_(), settings: readSettings_(), line: lineStatus_(), sheet: { name: ss.getName(), url: ss.getUrl() } };
      case 'save':
        return { ok: true, row: saveRow_(req.sheet, req.row) };
      case 'delete':
        deleteRow_(req.sheet, req.id);
        return { ok: true };
      case 'saveSettings':
        saveSettings_(req.settings || {});
        scheduleAll_();
        return { ok: true, settings: readSettings_() };
      case 'calendars':
        try { return { ok: true, calendars: listCalendars_() }; }
        catch (e) { return { ok: false, needAuth: true, error: 'Googleカレンダーを読む許可が、まだ出ていません。スプレッドシートの「Apps Script」で「カレンダーを許可する」を実行して、許可してください' }; }
      case 'saveLine':
        if (req.token) props.setProperty('LINE_TOKEN', String(req.token).trim());
        if (req.userId) props.setProperty('LINE_USER', String(req.userId).trim());
        return { ok: true, line: lineStatus_() };
      case 'changeKey':
        if (String(req.newKey || '').length < 4) return { ok: false, error: '新しいパスワードは4文字以上にしてください' };
        props.setProperty('KEY_HASH', hash_(String(req.newKey)));
        return { ok: true };
      case 'preview':
        // day：'today'＝今日の予定、'tomorrow'＝明日の予定。なし＝LINEに届くのと同じ（設定どおり）
        return { ok: true, text: buildFor_(req.day) };
      case 'testSend': {
        const q = lineQuota_();
        if (q && q.used >= TEST_STOP) return { ok: false, stopped: true, quota: q, error: '今月すでに' + TEST_STOP + '通に達しています（' + q.used + '通／' + (q.limit || 200) + '通）。毎日のリマインドのために残しておくため、テスト送信は止めています' };
        sendLine_(buildFor_());
        saveSettings_({ '準備：送信確認': '○' });
        return { ok: true, quota: lineQuota_(), settings: readSettings_() };
      }
      case 'quota':
        return { ok: true, quota: lineQuota_() };
      default:
        return { ok: false, error: '知らない操作です: ' + req.action };
    }
  } finally {
    lock.releaseLock();
  }
}

function hash_(s) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, 'line-hisho:' + s, Utilities.Charset.UTF_8);
  return bytes.map(b => ('0' + (b & 0xff).toString(16)).slice(-2)).join('');
}

function newId_() {
  return Utilities.getUuid().slice(0, 8);
}

// ───────── シートの読み書き ─────────
function readSheet_(name) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sh || sh.getLastRow() < 2) return [];
  const values = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getDisplayValues();
  const head = values[0];
  return values.slice(1).filter(r => r.some(v => v !== '')).map(r => {
    const o = {};
    head.forEach((h, i) => { o[h] = r[i]; });
    return o;
  });
}

function readAll_() {
  const out = {};
  Object.keys(SCHEMA).forEach(n => { out[n] = readSheet_(n); });
  return out;
}

function saveRow_(sheetName, row) {
  if (!SCHEMA[sheetName]) throw new Error('シート名がちがいます');
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  const head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  row = row || {};
  if (sheetName === 'タスク台帳') row['更新日'] = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
  if (!row.ID) row.ID = newId_();
  const line = head.map(h => (row[h] === undefined || row[h] === null) ? '' : String(row[h]));
  const ids = sh.getLastRow() >= 2 ? sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(r => String(r[0])) : [];
  const idx = ids.indexOf(String(row.ID));
  if (idx >= 0) {
    sh.getRange(idx + 2, 1, 1, head.length).setValues([line]);
  } else {
    sh.getRange(sh.getLastRow() + 1, 1, 1, head.length).setNumberFormat('@').setValues([line]);
  }
  const o = {};
  head.forEach((h, i) => { o[h] = line[i]; });
  return o;
}

function deleteRow_(sheetName, id) {
  if (!SCHEMA[sheetName]) throw new Error('シート名がちがいます');
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (sh.getLastRow() < 2) return;
  const ids = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(r => String(r[0]));
  const idx = ids.indexOf(String(id));
  if (idx >= 0) sh.deleteRow(idx + 2);
}

function readSettings_() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('設定');
  const o = {};
  if (!sh || sh.getLastRow() < 2) return o;
  sh.getRange(2, 1, sh.getLastRow() - 1, 2).getDisplayValues().forEach(r => { if (r[0]) o[r[0]] = r[1]; });
  return o;
}

function saveSettings_(s) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('設定');
  const rows = sh.getLastRow() >= 2 ? sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues() : [];
  Object.keys(SETTING_DEFAULTS).forEach(k => {
    if (!(k in s)) return;
    const v = String(s[k]);
    const i = rows.findIndex(r => r[0] === k);
    if (i >= 0) sh.getRange(i + 2, 2).setNumberFormat('@').setValue(v);
    else sh.appendRow([k, v]);
  });
}

// ───────── Googleカレンダー ─────────
function selectedCalendars_() {
  try { return JSON.parse(readSettings_()['カレンダー'] || '[]'); } catch (e) { return []; }
}

function listCalendars_() {
  const sel = selectedCalendars_();
  return CalendarApp.getAllCalendars().map(c => ({ id: c.getId(), name: c.getName(), selected: sel.indexOf(c.getId()) >= 0 }));
}

/** その日のカレンダーの予定（選んだカレンダーだけ） */
function calendarEvents_(dateStr) {
  const ids = selectedCalendars_();
  if (!ids.length) return [];
  const day = new Date(dateStr + 'T00:00:00+09:00');
  const out = [];
  ids.forEach(id => {
    const cal = CalendarApp.getCalendarById(id);
    if (!cal) return;
    cal.getEventsForDay(day).forEach(ev => {
      const s = ev.getStartTime(), e = ev.getEndTime();
      out.push({
        title: ev.getTitle(), place: ev.getLocation(), cal: cal.getName(), allDay: ev.isAllDayEvent(),
        sDate: Utilities.formatDate(s, TZ, 'yyyy-MM-dd'), sTime: Utilities.formatDate(s, TZ, 'HH:mm'),
        eDate: Utilities.formatDate(new Date(e.getTime() - (ev.isAllDayEvent() ? 1 : 0)), TZ, 'yyyy-MM-dd'), eTime: Utilities.formatDate(e, TZ, 'HH:mm')
      });
    });
  });
  return out;
}

// ───────── 毎日のリマインド（自動送信） ─────────
function 毎日のリマインド() { runNotice_(); }
// 前の版の予約が残っていたときのため
function 朝の通知() { runNotice_(); }
function 前夜の通知() { runNotice_(); }
function 見守り() { scheduleAll_(); } // 1日1回、予約が消えていないか確かめる

function runNotice_() {
  try {
    if (!licenseCheck_().ok) return; // 使えない状態なら送らない
    // 新しい版に入れ替えたあと、サイトを開く前に自動で送るときも、シートを今の形にそろえてから作る
    if (PropertiesService.getScriptProperties().getProperty('SCHEMA_VER') !== VERSION) ensureAll_();
    let text = buildFor_();
    try {
      const q = lineQuota_();
      if (q && q.used >= TEST_STOP) text += '\n\n※今月のLINE送信数が' + TEST_STOP + '通に達しました（' + q.used + '通／' + (q.limit || 200) + '通）。';
    } catch (e) {}
    sendLine_(text);
  } finally {
    scheduleAll_();
  }
}

function scheduleAll_() {
  const s = Object.assign({}, SETTING_DEFAULTS, readSettings_());
  const handlers = ['毎日のリマインド', '朝の通知', '前夜の通知', '見守り'];
  ScriptApp.getProjectTriggers().forEach(t => {
    if (handlers.indexOf(t.getHandlerFunction()) >= 0) ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('毎日のリマインド').timeBased().at(nextAt_(s['リマインドの時刻'])).create();
  ScriptApp.newTrigger('見守り').timeBased().everyDays(1).atHour(3).create();
}

/** 次に来る「HH:mm」（日本時間）の日時 */
function nextAt_(hhmm) {
  const m = /^(\d{1,2}):(\d{2})$/.exec(String(hhmm || '').trim()) || [null, '7', '00'];
  const hh = ('0' + m[1]).slice(-2), mm = m[2];
  const now = new Date();
  let day = Utilities.formatDate(now, TZ, 'yyyy-MM-dd');
  let t = new Date(day + 'T' + hh + ':' + mm + ':00+09:00');
  if (t.getTime() <= now.getTime() + 60 * 1000) {
    t = new Date(t.getTime() + 24 * 60 * 60 * 1000);
  }
  return t;
}

function buildFor_(day) {
  const s = Object.assign({}, SETTING_DEFAULTS, readSettings_());
  const next = day === 'today' ? false : day === 'tomorrow' ? true : s['翌日の予定を送る'] === '○';
  const now = new Date();
  const target = next ? new Date(now.getTime() + 24 * 60 * 60 * 1000) : now;
  const dateStr = Utilities.formatDate(target, TZ, 'yyyy-MM-dd');
  let cal = [];
  try { cal = calendarEvents_(dateStr); } catch (e) { cal = [{ error: true }]; }
  return buildMessage(readAll_(), s, dateStr, next ? 'next' : 'today', cal);
}

// ───────── 日付の計算（ここはシートに触らない） ─────────
const WEEK = ['日', '月', '火', '水', '木', '金', '土'];

function ymd_(s) { const p = String(s || '').split('-').map(Number); return (p.length === 3 && p[0]) ? p : null; }
function toNum_(s) { const p = ymd_(s); return p ? Date.UTC(p[0], p[1] - 1, p[2]) / 86400000 : null; }
function fromNum_(n) { const d = new Date(n * 86400000); return d.getUTCFullYear() + '-' + ('0' + (d.getUTCMonth() + 1)).slice(-2) + '-' + ('0' + d.getUTCDate()).slice(-2); }
function wdOf_(s) { return WEEK[new Date(toNum_(s) * 86400000).getUTCDay()]; }
function md_(s) { const p = ymd_(s); return p ? p[1] + '/' + p[2] + '（' + wdOf_(s) + '）' : ''; }

/** 何か月前（同じ日付。ない日は月末にそろえる） */
function monthsBefore_(s, k) {
  const p = ymd_(s);
  let y = p[0], m = p[1] - k;
  while (m < 1) { m += 12; y--; }
  const last = new Date(Date.UTC(y, m, 0)).getUTCDate();
  return y + '-' + ('0' + m).slice(-2) + '-' + ('0' + Math.min(p[2], last)).slice(-2);
}

/** リマインドの選択肢 → 予定の日から、知らせる日を出す */
function offsetDate_(eventDate, label) {
  const m = /^(\d+)か月前$/.exec(label);
  if (m) return monthsBefore_(eventDate, Number(m[1]));
  const w = /^(\d+)週間前$/.exec(label);
  if (w) return fromNum_(toNum_(eventDate) - 7 * Number(w[1]));
  const d = /^(\d+)日前$/.exec(label);
  if (d) return fromNum_(toNum_(eventDate) - Number(d[1]));
  if (label === '前日') return fromNum_(toNum_(eventDate) - 1);
  return null;
}

/** 定例が、その日に始まるか（Googleカレンダーの「カスタム」と同じ考え方） */
function occursOn_(r, s) {
  const n = toNum_(s), p = ymd_(s), wd = wdOf_(s);
  const base = toNum_(r['開始日']), endN = toNum_(r['終了日']);
  if (base !== null && n < base) return false;
  if (endN !== null && n > endN) return false;
  const k = Math.max(1, Math.floor(Number(r['間隔']) || 1));
  const step = (diff) => base === null ? k === 1 : diff % k === 0;
  switch (r['単位']) {
    case '日':
      return step(n - (base === null ? 0 : base));
    case '週': {
      const days = String(r['曜日'] || '') || (base !== null ? wdOf_(r['開始日']) : '');
      if (days.indexOf(wd) < 0) return false;
      const wk = x => Math.floor((x + 3) / 7); // 月曜はじまりの週の番号
      return step(wk(n) - (base === null ? 0 : wk(base)));
    }
    case 'か月': {
      const bp = ymd_(r['開始日']);
      if (!step(p[0] * 12 + p[1] - (bp ? bp[0] * 12 + bp[1] : 0))) return false;
      const last = new Date(Date.UTC(p[0], p[1], 0)).getUTCDate();
      if ((r['月の決め方'] || '日付で') === '日付で') return r['日'] === String(p[2]) || (r['日'] === '末日' && p[2] === last);
      if (r['何曜日'] !== wd) return false;
      if (r['第何週'] === '最終週') return p[2] + 7 > last;
      const w = Number(String(r['第何週'] || '').replace(/[^0-9]/g, ''));
      return w >= 1 && p[2] > (w - 1) * 7 && p[2] <= w * 7;
    }
    case '年': {
      const bp = ymd_(r['開始日']);
      if (!bp || bp[1] !== p[1] || bp[2] !== p[2]) return false;
      return (p[0] - bp[0]) % k === 0;
    }
  }
  return false;
}

function spanDays_(r) { const n = Number(r['何日間'] || 1); return n >= 1 && n <= 31 ? Math.floor(n) : 1; }

/** 予定1つ分の文字（何日目かも付ける） */
function eventLine_(o, dateStr) {
  const n = toNum_(dateStr), sN = toNum_(o.sDate), eN = toNum_(o.eDate || o.sDate);
  const where = o.place ? '（' + o.place + '）' : '';
  const tag = o.tag ? '［' + o.tag + '］' : '';
  let time;
  if (eN > sN) {
    if (n === sN) time = (o.allDay || !o.sTime ? '' : o.sTime) + '〜' + md_(o.eDate) + (o.allDay || !o.eTime ? '' : ' ' + o.eTime) + 'まで';
    else if (n === eN) time = '最終日' + (o.allDay || !o.eTime ? '' : '〜' + o.eTime);
    else time = '続き（' + (n - sN + 1) + '日目）';
  } else if (o.allDay || (!o.sTime && !o.eTime)) {
    time = o.allDay ? '終日' : '';
  } else {
    time = (o.sTime || '') + (o.eTime ? '〜' + o.eTime : '');
  }
  return { sort: (n === sN && o.sTime && !o.allDay) ? o.sTime : (n === sN ? '00:00' : '00:01'), text: '・' + (time ? time + ' ' : '') + o.title + where + tag };
}

// ───────── メッセージを組み立てる（ここはシートに触らない） ─────────
function buildMessage(data, settings, dateStr, kind, calEvents) {
  const p = ymd_(dateStr), mo = p[1], d = p[2], wd = wdOf_(dateStr), today = toNum_(dateStr);
  const on = v => String(v || '') === '○';
  const cat = c => c ? '［' + c + '］' : '';

  const irr = (data['イレギュラー'] || []);
  const irrToday = irr.filter(r => r['日付'] === dateStr);
  const dayOff = irrToday.some(r => r['種類'] === 'この日は定例をお休み');
  const stopped = irrToday.filter(r => r['種類'] === 'この日だけ定例を止める').map(r => String(r['内容'] || '').trim());
  const notStopped = r => stopped.indexOf(String(r['内容'] || '').trim()) < 0;

  // ── 予定（カレンダー・定例・単発） ──
  const events = [];
  (calEvents || []).filter(e => !e.error).forEach(e => events.push({ title: e.title, place: e.place, tag: e.cal, allDay: e.allDay, sDate: e.sDate, sTime: e.sTime, eDate: e.eDate, eTime: e.eTime }));
  const blocks = [];
  if (!dayOff) {
    (data['定例'] || []).filter(r => on(r['有効']) && r['内容']).filter(notStopped).forEach(r => {
      const n = spanDays_(r);
      for (let back = 0; back < n; back++) {
        const start = fromNum_(today - back);
        if (occursOn_(r, start)) {
          const o = { title: r['内容'], place: r['場所'], tag: r['カテゴリ'], sDate: start, sTime: r['開始'], eDate: fromNum_(toNum_(start) + n - 1), eTime: r['終了'] };
          if (r['種類'] === '時間の確保') { const x = eventLine_(o, dateStr); blocks.push({ sort: r['開始'] || '99:99', text: x.text }); }
          else events.push(o);
          break;
        }
      }
    });
  }
  irr.filter(r => r['種類'] === '予定' && r['内容']).forEach(r => {
    const s = toNum_(r['日付']), e = toNum_(r['終了日']) || s;
    if (s !== null && today >= s && today <= e) events.push({ title: r['内容'], place: r['場所'], tag: r['カテゴリ'], sDate: r['日付'], sTime: r['開始'], eDate: r['終了日'] || r['日付'], eTime: r['終了'] });
  });
  const lines = events.map(o => eventLine_(o, dateStr));

  const bySort = (a, b) => a.sort.localeCompare(b.sort);
  lines.sort(bySort); blocks.sort(bySort);

  // ── 先の予定のリマインド（リマインドチェック） ──
  const ahead = [];
  const checkAhead = (title, eventDate, offsets, tagText) => {
    String(offsets || '').split(',').map(x => x.trim()).filter(Boolean).forEach(lb => {
      if (offsetDate_(eventDate, lb) === dateStr) {
        const left = toNum_(eventDate) - today;
        const say = lb === '前日' ? 'いよいよ明日です' : 'あと' + lb.replace(/前$/, '') + 'です' + (/か月|週間/.test(lb) ? '（' + left + '日）' : '');
        ahead.push({ n: left, text: '・' + title + '（' + md_(eventDate) + '）まで、' + say + tagText });
      }
    });
  };
  for (let k = 1; k <= 100; k++) {
    const day = fromNum_(today + k);
    (data['定例'] || []).filter(r => on(r['有効']) && r['リマインド']).forEach(r => {
      if (occursOn_(r, day)) checkAhead(r['内容'], day, r['リマインド'], cat(r['カテゴリ']));
    });
  }
  irr.filter(r => r['種類'] === '予定' && r['リマインド'] && toNum_(r['日付']) > today).forEach(r => checkAhead(r['内容'], r['日付'], r['リマインド'], cat(r['カテゴリ'])));

  // ── タスク ──
  const tasks = (data['タスク台帳'] || []).filter(r => r['状態'] !== '完了');
  tasks.filter(r => r['リマインド'] && toNum_(r['期限']) > today).forEach(r => checkAhead('【期限】' + r['タスク'], r['期限'], r['リマインド'], cat(r['カテゴリ'])));
  ahead.sort((a, b) => a.n - b.n);
  const urgent = tasks.filter(r => on(r['緊急']));
  const diff = s => { const n = toNum_(s); return n === null ? null : n - today; };
  const due = tasks.filter(r => !on(r['緊急'])).filter(r => { const n = diff(r['期限']); return n !== null && n <= 1; })
    .sort((a, b) => diff(a['期限']) - diff(b['期限']));
  const dueIds = due.map(r => r.ID);
  const rest = tasks.filter(r => !on(r['緊急']) && dueIds.indexOf(r.ID) < 0);
  const stuck = rest.filter(r => r['状態'] === '止まっている' || r['状態'] === 'お返事待ち');
  const doing = rest.filter(r => r['状態'] === '途中');
  const taskLine = r => '・' + r['タスク'] + cat(r['カテゴリ']) + (r['次にやること'] ? '\n　→ 次：' + r['次にやること'] : '');
  const pmd = s => { const q = ymd_(s); return q ? q[1] + '/' + q[2] : ''; };

  const name = settings['秘書の名前'] || '秘書';
  const head = kind === 'next'
    ? 'お疲れさまです、' + name + 'です。\n明日 ' + mo + '月' + d + '日（' + wd + '）の予定です。'
    : 'おはようございます、' + name + 'です。\n今日 ' + mo + '月' + d + '日（' + wd + '）の予定です。';
  const parts = [head];
  if (dayOff) parts.push('この日は定例を「お休み」にしています。');
  if (urgent.length) parts.push('【緊急】\n' + urgent.map(r => taskLine(r) + (r['期限'] ? '（期限 ' + pmd(r['期限']) + '）' : '')).join('\n'));
  if (lines.length) parts.push('【' + (kind === 'next' ? '明日' : '今日') + 'の予定】\n' + lines.map(x => x.text).join('\n'));
  if (blocks.length) parts.push('【時間の確保】\n' + blocks.map(x => x.text).join('\n'));
  if ((calEvents || []).some(e => e.error)) parts.push('※ Googleカレンダーを読めませんでした。設定の「カレンダーの連携」を確かめてください。');
  if (ahead.length) parts.push('【先の予定の準備】\n' + ahead.map(x => x.text).join('\n'));
  if (due.length) parts.push('【期限が近い】\n' + due.map(r => {
    const n = diff(r['期限']);
    const label = n < 0 ? '期限切れ ' + pmd(r['期限']) : n === 0 ? '今日が期限' : '明日が期限';
    return '・' + label + ' ' + r['タスク'] + cat(r['カテゴリ']);
  }).join('\n'));
  if (stuck.length) parts.push('【止まっている・お返事待ち】\n' + stuck.map(r => '・' + r['タスク'] + '（' + r['状態'] + '）' + cat(r['カテゴリ'])).join('\n'));
  if (doing.length) parts.push('【進行中】\n' + doing.map(taskLine).join('\n'));
  if (parts.length === 1) parts.push('お知らせすることは登録されていません。');

  let text = parts.join('\n\n');
  if (text.length > 4900) text = text.slice(0, 4880) + '\n…（長いので省略しました）';
  return text;
}

// ───────── 利用の確認 ─────────
/** 顧客管理に「このスプレッドシートを使ってよいか」を聞く（結果は1時間覚えておく） */
function licenseCheck_() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('LIC');
  if (hit === 'ok') return { ok: true };
  const acc = PropertiesService.getScriptProperties().getProperty('ACCOUNT_ID');
  if (!acc) return { ok: false, error: 'まだ利用者IDとつながっていません。設定サイトからログインしてつないでください' };
  let j;
  try {
    const res = UrlFetchApp.fetch(LICENSE_URL, {
      method: 'post', contentType: 'text/plain', muteHttpExceptions: true, followRedirects: true,
      payload: JSON.stringify({ action: 'check', id: acc, sheetId: SpreadsheetApp.getActiveSpreadsheet().getId() })
    });
    j = JSON.parse(res.getContentText());
  } catch (e) {
    return { ok: false, error: '利用の確認ができませんでした。少し時間をおいてやり直してください' };
  }
  if (j && j.ok) { cache.put('LIC', 'ok', 3600); return { ok: true }; }
  return { ok: false, error: (j && j.error) || '利用の確認ができませんでした' };
}

/** 別の利用者IDにつなぎ直したいとき：エディタからこれを実行すると、利用者IDとのひもづけを外します */
function 利用者IDのひもづけを外す() {
  PropertiesService.getScriptProperties().deleteProperty('ACCOUNT_ID');
  CacheService.getScriptCache().remove('LIC');
  Logger.log('利用者IDとのひもづけを外しました。');
}

// ───────── LINE ─────────
function lineStatus_() {
  const p = PropertiesService.getScriptProperties();
  return { token: !!p.getProperty('LINE_TOKEN'), userId: !!p.getProperty('LINE_USER') };
}

function sendLine_(text) {
  const p = PropertiesService.getScriptProperties();
  const token = p.getProperty('LINE_TOKEN'), user = p.getProperty('LINE_USER');
  if (!token || !user) throw new Error('LINEの接続情報がまだ入っていません');
  const res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ to: user, messages: [{ type: 'text', text: text }] }),
    muteHttpExceptions: true
  });
  if (res.getResponseCode() !== 200) {
    throw new Error('LINEに送れませんでした（' + res.getResponseCode() + '）：' + res.getContentText());
  }
}

/** 今月送れる数と、使った数 */
function lineQuota_() {
  const token = PropertiesService.getScriptProperties().getProperty('LINE_TOKEN');
  if (!token) return null;
  const opt = { headers: { Authorization: 'Bearer ' + token }, muteHttpExceptions: true };
  const q = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/quota', opt);
  const c = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/quota/consumption', opt);
  if (q.getResponseCode() !== 200 || c.getResponseCode() !== 200) return null;
  const qj = JSON.parse(q.getContentText()), cj = JSON.parse(c.getContentText());
  return { limit: qj.type === 'limited' ? qj.value : null, used: cj.totalUsage };
}
